P=/c/Nyasha_Planet/LMX/build2/codex_fix250generationchain01
SP=/c/Users/mtkra/AppData/Local/Temp/claude/6d92abb4660ab27f/b79b4933-abc1-4eae-98f7-6361633964ee/scratchpad
PWSH="/c/Users/mtkra/.cache/codex-runtimes/codex-primary-runtime/dependencies/native/powershell/pwsh.exe"
cd $P
run() { "$PWSH" -NoProfile -Command "& ([ScriptBlock]::Create([IO.File]::ReadAllText('C:/Nyasha_Planet/LMX/build2/codex_fix250generationchain01/pins/$1.command')))" < /dev/null 2>&1 | tail -n 1 | cut -c1-200; }
OUT=$P/pins/mut280.txt
TOOL=cold-tp140/tool142a/lmxtool-tool.exe
BASEFE=cold-tp140/fe142d/l2trans.exe
BASERT=cold-tp140/rt142b
n=$1; shift
for m in "$@"; do
  n=$((n+1))
  if ! python -I $SP/mut280.py $m >> $OUT 2>&1; then echo "mutation did not apply: $m" >> $OUT; continue; fi
  FE=$BASEFE
  RT=$BASERT
  case "$m" in
    kernel*)
      TP_TOOL=$TOOL TP_LABEL=rtm142-$m TP_TRY=m$m run tp142-rt >> $OUT
      RT=cold-tp140/rtm142-$m
      ;;
    *)
      TP_TOOL=$TOOL TP_LABEL=fem142-$m TP_TRY=f$m run tp142-fe >> $OUT
      FE=cold-tp140/fem142-$m/l2trans.exe
      ;;
  esac
  TP_TOOL=$TOOL TP_FE=$FE TP_RT=$RT TP_TRY=f$m TP_LIST=probe280-p.tsv run tp142-probe >> $OUT
  grep -a -E '^ *SUITE' pins/tp142-probe-f$m.terminal.txt | cut -c1-80 >> $OUT
  for k in native forced; do
    printf '%s %s: ' $m $k >> $OUT
    { grep -a -h 'l2trans error' tp142-probe/f$m/structure_slot_write-$k/p.L2.log 2>/dev/null | head -n 1 | sed 's|.*/tests/||' | cut -c1-110; for r in 0 1; do grep -a -h -E 'exit [0-9]+, expected|walk error|invariant|R0 was stopped' tp142-probe/f$m/structure_slot_write-$k/run-root$r.log 2>/dev/null | head -n 1 | cut -c1-70; done; } | tr '\n' ';' >> $OUT
    echo >> $OUT
  done
  # the stage comes back to the candidate through its own recipe; failure stops the loop so no mutation carries over
  r=$(TP_TRY=m$n run tp142-prepare)
  echo "$m restore: $r" >> $OUT
  case "$r" in PREPARED*) ;; *) echo "restore failed, stopped" >> $OUT; exit 1;; esac
done
echo done
